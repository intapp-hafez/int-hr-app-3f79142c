# ==============================================================================
# INT-HR App — Apache / LiteSpeed / cPanel / hPanel Configuration
# NOTE: If your server does not see .htaccess (hidden file), rename this file
# from htaccess.txt to .htaccess in your cPanel/hPanel File Manager!
# ==============================================================================

# Disable MultiViews so Apache doesn't misinterpret client routes (e.g. /admin, /reports)
Options -MultiViews
Options +FollowSymLinks

# DirectoryIndex ensure index.html is the default document
<IfModule mod_dir.c>
  DirectoryIndex index.html
</IfModule>

# ─── 1. HTTPS Enforcement & SPA Routing ───────────────────────────────────────
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # Force HTTPS if connection is not secure (supports Direct, Cloudflare, and Reverse Proxy)
  RewriteCond %{HTTPS} off
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

  # If the requested resource is an existing file or directory, serve directly
  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]

  # SPA Routing: Route all client routes (e.g. /admin/shifts, /login) to index.html
  RewriteRule ^ index.html [L]
</IfModule>

# ─── 2. Fallback Safety Net for 404 On Reload ─────────────────────────────────
# If mod_rewrite is bypassed or encounters an error, serve index.html so React Router takes over
ErrorDocument 404 /index.html

# ─── 3. HTTP Security Headers (Fixes Security Scanner Grade F -> A+) ──────────
<IfModule mod_headers.c>
  # 1. HTTP Strict Transport Security (HSTS)
  Header set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"

  # 2. X-Frame-Options (Clickjacking protection)
  Header set X-Frame-Options "SAMEORIGIN"
  Header always set X-Frame-Options "SAMEORIGIN"

  # 3. X-Content-Type-Options (MIME-type sniffing protection)
  Header set X-Content-Type-Options "nosniff"
  Header always set X-Content-Type-Options "nosniff"

  # 4. X-XSS-Protection (Legacy browser protection)
  Header set X-XSS-Protection "1; mode=block"
  Header always set X-XSS-Protection "1; mode=block"

  # 5. Referrer-Policy
  Header set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

  # 6. Permissions-Policy: Enable GPS Geolocation & Camera for employee attendance & selfie check-in
  Header set Permissions-Policy "geolocation=(self), camera=(self), microphone=(), payment=(), usb=()"
  Header always set Permissions-Policy "geolocation=(self), camera=(self), microphone=(), payment=()"

  # 7. Content-Security-Policy (CSP)
  Header set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval'; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com data:; img-src 'self' data: blob: https://*.supabase.co https://*.tile.openstreetmap.org https://tile.openstreetmap.org; connect-src 'self' https://*.supabase.co wss://*.supabase.co https://*.tile.openstreetmap.org https://tile.openstreetmap.org https://nominatim.openstreetmap.org https://fonts.googleapis.com https://fonts.gstatic.com; worker-src 'self' blob:; frame-ancestors 'self';"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval'; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com data:; img-src 'self' data: blob: https://*.supabase.co https://*.tile.openstreetmap.org https://tile.openstreetmap.org; connect-src 'self' https://*.supabase.co wss://*.supabase.co https://*.tile.openstreetmap.org https://tile.openstreetmap.org https://nominatim.openstreetmap.org https://fonts.googleapis.com https://fonts.gstatic.com; worker-src 'self' blob:; frame-ancestors 'self';"

  # Remove server technology fingerprints
  Header unset X-Powered-By
  Header always unset X-Powered-By
</IfModule>

# ─── 4. MIME Types for Modern Web, PWA & Fonts ────────────────────────────────
<IfModule mod_mime.c>
  AddType application/manifest+json .webmanifest
  AddType application/json .json
  AddType font/woff2 .woff2
  AddType font/woff .woff
  AddType text/javascript .mjs .js
  AddType image/svg+xml .svg
</IfModule>

# ─── 5. Cache-Control Optimization ────────────────────────────────────────────
<IfModule mod_headers.c>
  # Never cache index.html, service worker, or manifest so releases apply immediately
  <FilesMatch "(index\.html|sw\.js|sw-push\.js|app\.webmanifest|manifest\.json)$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
    Header set Pragma "no-cache"
    Header set Expires "0"
  </FilesMatch>

  # Cache hashed static assets (JS, CSS, images, fonts) for 1 year
  <FilesMatch "\.(css|js|mjs|woff2|woff|png|jpg|jpeg|svg|webp|ico)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

# ─── 6. Compression (Gzip / Deflate) ──────────────────────────────────────────
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css text/javascript application/javascript application/json application/manifest+json image/svg+xml
</IfModule>
